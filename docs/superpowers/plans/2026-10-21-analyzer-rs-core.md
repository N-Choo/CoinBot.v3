# Analyzer-RS Phase 1 — Pure Core + Python Parity Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Create the `process/analyzer-rs` crate and port the Python RSI-divergence strategy into a pure `Decimal`-only core that matches Python on indicators, divergences, and signals.

**Architecture:** A new workspace crate with a single `strategy` module. Every indicator and the signal generator are pure functions over a candle slice; no I/O, no clock, no floats. Python artifacts generated by `process/analyzer/tools/parity/emit.py` are the golden oracle.

**Tech Stack:** Rust 2021, `rust_decimal`, `serde`, `thiserror`; dev-only `syn`/`proc-macro2` for the purity test, `serde_json` for fixtures. Python 3 with the existing `process/analyzer` package (`pandas`, `scipy`) generates fixtures.

**Scope:** This plan delivers Phase 1 only (core + parity). The driver/backtest/HTML plan and the live/gRPC plan are follow-on plans.

```rust
use crate::strategy::error::{Error, Result};
use rust_decimal::Decimal;

pub fn add(a: Decimal, b: Decimal) -> Result<Decimal> {
    a.checked_add(b).ok_or(Error::Overflow)
}

pub fn sub(a: Decimal, b: Decimal) -> Result<Decimal> {
    a.checked_sub(b).ok_or(Error::Overflow)
}

pub fn mul(a: Decimal, b: Decimal) -> Result<Decimal> {
    a.checked_mul(b).ok_or(Error::Overflow)
}

pub fn div(a: Decimal, b: Decimal) -> Result<Decimal> {
    if b == Decimal::ZERO {
        return Err(Error::DivByZero);
    }
    a.checked_div(b).ok_or(Error::Overflow)
}

pub fn neg(a: Decimal) -> Result<Decimal> {
    Decimal::ZERO.checked_sub(a).ok_or(Error::Overflow)
}

pub fn abs(a: Decimal) -> Result<Decimal> {
    if a < Decimal::ZERO {
        neg(a)
    } else {
        Ok(a)
    }
}
```

```rust
use crate::strategy::error::{Error, Result};
use rust_decimal::Decimal;
use std::str::FromStr;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub rsi_period: usize,
    pub rsi_order: usize,
    pub price_diff_threshold: Decimal,
    pub sma_enabled: bool,
    pub sma_period: usize,
    pub atr_period: usize,
    pub atr_sl_multiplier: Decimal,
    pub atr_tp_multiplier: Decimal,
    pub atr_pct_high: Decimal,
    pub atr_high_adjust: Decimal,
    pub atr_sl_max_pct: Decimal,
    pub window: usize,
}

impl Default for Config {
    fn default() -> Self {
        Config {
            rsi_period: 14,
            rsi_order: 5,
            price_diff_threshold: Decimal::new(5, 2),
            sma_enabled: false,
            sma_period: 20,
            atr_period: 14,
            atr_sl_multiplier: Decimal::new(20, 1),
            atr_tp_multiplier: Decimal::new(30, 1),
            atr_pct_high: Decimal::new(50, 1),
            atr_high_adjust: Decimal::new(15, 1),
            atr_sl_max_pct: Decimal::ZERO,
            window: 100,
        }
    }
}

impl Config {
    pub fn min_window(&self) -> usize {
        let sma = if self.sma_enabled { self.sma_period } else { 0 };
        self.rsi_period
            .max(self.atr_period)
            .max(sma)
            .saturating_add(self.rsi_order)
            .saturating_add(2)
    }

    pub fn validate(&self) -> Result<()> {
        let fail = |msg: &str| Err(Error::ConfigInvalid(msg.to_string()));
        if self.rsi_period == 0 {
            return fail("rsi_period must be >= 1");
        }
        if self.rsi_order == 0 {
            return fail("rsi_order must be >= 1");
        }
        if self.sma_period == 0 {
            return fail("sma_period must be >= 1");
        }
        if self.atr_period == 0 {
            return fail("atr_period must be >= 1");
        }
        if self.price_diff_threshold < Decimal::ZERO {
            return fail("price_diff_threshold must be >= 0");
        }
        if self.atr_sl_multiplier <= Decimal::ZERO || self.atr_tp_multiplier <= Decimal::ZERO {
            return fail("atr multipliers must be > 0");
        }
        if self.atr_pct_high < Decimal::ZERO {
            return fail("atr_pct_high must be >= 0");
        }
        if self.atr_high_adjust <= Decimal::ZERO {
            return fail("atr_high_adjust must be > 0");
        }
        if self.atr_sl_max_pct < Decimal::ZERO || self.atr_sl_max_pct >= Decimal::ONE {
            return fail("atr_sl_max_pct must be in [0, 1)");
        }
        if self.window < self.min_window() {
            return fail("window is below the indicator minimum");
        }
        if self.window < 8 {
            return fail("window must be >= 8");
        }
        Ok(())
    }
}

pub fn decimal_from_str(s: &str) -> Result<Decimal> {
    Decimal::from_str(s).map_err(|_| Error::ConfigInvalid(format!("invalid decimal {s:?}")))
}
```

```rust
use crate::strategy::error::{Error, Result};
use crate::strategy::types::Candle;

#[derive(Debug, Clone)]
pub struct Window {
    candles: Vec<Candle>,
    capacity: usize,
}

impl Window {
    pub fn new(capacity: usize) -> Self {
        Window {
            candles: Vec::with_capacity(capacity),
            capacity,
        }
    }

    pub fn push_closed(&mut self, candle: Candle) -> Result<()> {
        if self.capacity == 0 {
            return Err(Error::InvalidWindowCapacity);
        }
        if let Some(last) = self.candles.last() {
            if candle.open_time_ms == last.open_time_ms {
                return Err(Error::DuplicateCandle(candle.open_time_ms));
            }
            if candle.open_time_ms < last.open_time_ms {
                return Err(Error::OutOfOrderCandle {
                    got_ms: candle.open_time_ms,
                    last_ms: last.open_time_ms,
                });
            }
        }
        if self.candles.len() == self.capacity {
            self.candles.remove(0);
        }
        self.candles.push(candle);
        Ok(())
    }

    pub fn as_slice(&self) -> &[Candle] {
        &self.candles
    }

    pub fn len(&self) -> usize {
        self.candles.len()
    }

    pub fn is_full(&self) -> bool {
        self.candles.len() == self.capacity
    }
}
```

```rust
use crate::strategy::decimal::{add, div, mul, sub};
use crate::strategy::error::{Error, Result};
use rust_decimal::Decimal;

pub mod atr;
pub mod peaks;
pub mod rsi;
pub mod sma;

pub type Series = Vec<Option<Decimal>>;

/// pandas `ewm(alpha=1/period, adjust=False)` with `skipna=True`.
pub fn ewm(values: &[Option<Decimal>], period: usize) -> Result<Series> {
    if period == 0 {
        return Err(Error::ConfigInvalid("ewm period must be >= 1".into()));
    }
    let alpha = div(Decimal::ONE, Decimal::from(period))?;
    let one_minus = sub(Decimal::ONE, alpha)?;
    let mut out = vec![None; values.len()];
    let mut prev: Option<Decimal> = None;
    for (i, value) in values.iter().enumerate() {
        match (value, prev) {
            (Some(x), Some(p)) => {
                let next = add(mul(alpha, *x)?, mul(one_minus, p)?)?;
                out[i] = Some(next);
                prev = Some(next);
            }
            (Some(x), None) => {
                out[i] = Some(*x);
                prev = Some(*x);
            }
            (None, _) => {}
        }
    }
    Ok(out)
}
```

```rust
use crate::strategy::decimal::{add, div};
use crate::strategy::error::{Error, Result};
use crate::strategy::indicators::Series;
use rust_decimal::Decimal;

pub fn sma(values: &[Option<Decimal>], period: usize) -> Result<Series> {
    if period == 0 {
        return Err(Error::ConfigInvalid("sma period must be >= 1".into()));
    }
    let mut out = vec![None; values.len()];
    if values.len() < period {
        return Ok(out);
    }
    for i in (period - 1)..values.len() {
        let mut sum = Decimal::ZERO;
        let mut valid = true;
        for value in &values[i + 1 - period..=i] {
            match value {
                Some(v) => sum = add(sum, *v)?,
                None => {
                    valid = false;
                    break;
                }
            }
        }
        if valid {
            out[i] = Some(div(sum, Decimal::from(period))?);
        }
    }
    Ok(out)
}
```

```rust
use crate::strategy::decimal::{add, div, neg, sub};
use crate::strategy::error::{Error, Result};
use crate::strategy::indicators::{ewm, Series};
use rust_decimal::Decimal;

pub fn rsi(closes: &[Option<Decimal>], period: usize) -> Result<Series> {
    if period == 0 {
        return Err(Error::ConfigInvalid("rsi period must be >= 1".into()));
    }
    let mut delta = vec![None; closes.len()];
    for i in 1..closes.len() {
        if let (Some(now), Some(prev)) = (closes[i], closes[i - 1]) {
            delta[i] = Some(sub(now, prev)?);
        }
    }

    let mut gain = vec![Some(Decimal::ZERO); closes.len()];
    let mut loss = vec![Some(Decimal::ZERO); closes.len()];
    for (i, d) in delta.iter().enumerate() {
        if let Some(d) = d {
            if *d > Decimal::ZERO {
                gain[i] = Some(*d);
            } else if *d < Decimal::ZERO {
                loss[i] = Some(neg(*d)?);
            }
        }
    }

    let avg_gain = ewm(&gain, period)?;
    let avg_loss = ewm(&loss, period)?;
    let epsilon = Decimal::new(1, 10);
    let hundred = Decimal::from(100);
    let mut out = vec![None; closes.len()];
    for i in 0..closes.len() {
        if let (Some(g), Some(l)) = (avg_gain[i], avg_loss[i]) {
            let rs = div(g, add(l, epsilon)?)?;
            let ratio = div(hundred, add(Decimal::ONE, rs)?)?;
            out[i] = Some(sub(hundred, ratio)?);
        }
    }
    Ok(out)
}
```

```rust
pub fn atr(candles: &[Candle], period: usize) -> Result<Series> {
    if period == 0 {
        return Err(Error::ConfigInvalid("atr period must be >= 1".into()));
    }
    let mut true_range = vec![None; candles.len()];
    for (i, candle) in candles.iter().enumerate() {
        let high_low = sub(candle.high, candle.low)?;
        if i == 0 {
            true_range[i] = Some(high_low);
        } else {
            let prev = candles[i - 1].close;
            let high_close = abs(sub(candle.high, prev)?)?;
            let low_close = abs(sub(candle.low, prev)?)?;
            let mut max = high_low;
            if high_close > max {
                max = high_close;
            }
            if low_close > max {
                max = low_close;
            }
            true_range[i] = Some(max);
        }
    }
    ewm(&true_range, period)
}
```

```rust
use rust_decimal::Decimal;

/// A `Decimal`/`Option` port of `scipy.signal.find_peaks(values, distance=distance)`.
/// `None` behaves like NaN: it never forms a peak and never blocks one.
pub fn find_peaks(values: &[Option<Decimal>], distance: usize) -> Vec<usize> {
    let peaks = local_maxima(values);
    if distance <= 1 {
        return peaks;
    }
    select_by_distance(&peaks, values, distance)
}

fn local_maxima(values: &[Option<Decimal>]) -> Vec<usize> {
    let n = values.len();
    let mut peaks = Vec::new();
    let mut i = 1usize;
    while i + 1 < n {
        let (prev, cur, next) = (values[i - 1], values[i], values[i + 1]);
        let (Some(prev), Some(cur), Some(next)) = (prev, cur, next) else {
            i += 1;
            continue;
        };
        if cur > prev && cur > next {
            peaks.push(i);
            i += 1;
        } else if cur > prev && cur == next {
            let mut j = i + 1;
            while j + 1 < n && values[j] == Some(cur) && values[j + 1] == Some(cur) {
                j += 1;
            }
            if let Some(after) = values.get(j + 1).copied().flatten() {
                if after < cur {
                    peaks.push(i + (j - i) / 2);
                }
            }
            i = j + 1;
        } else {
            i += 1;
        }
    }
    peaks
}

fn select_by_distance(peaks: &[usize], values: &[Option<Decimal>], distance: usize) -> Vec<usize> {
    use std::cmp::Ordering;
    let mut order: Vec<usize> = (0..peaks.len()).collect();
    order.sort_by(|&a, &b| {
        let va = values[peaks[a]].unwrap_or(Decimal::ZERO);
        let vb = values[peaks[b]].unwrap_or(Decimal::ZERO);
        match va.cmp(&vb) {
            Ordering::Less => Ordering::Greater,
            Ordering::Greater => Ordering::Less,
            Ordering::Equal => peaks[b].cmp(&peaks[a]),
        }
    });

    let mut kept: Vec<usize> = Vec::new();
    for &index in &order {
        let position = peaks[index];
        if kept
            .iter()
            .all(|&k| position.abs_diff(peaks[k]) >= distance)
        {
            kept.push(index);
        }
    }

    let mut result: Vec<usize> = kept.into_iter().map(|i| peaks[i]).collect();
    result.sort_unstable();
    result
}
```

```rust
use crate::strategy::error::Result;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Divergences {
    pub rsi: Vec<Option<Decimal>>,
    pub regular_bullish: Vec<bool>,
    pub regular_bearish: Vec<bool>,
    pub hidden_bullish: Vec<bool>,
    pub hidden_bearish: Vec<bool>,

pub fn detect(candles: &[Candle], cfg: &Config) -> Result<Divergences> {
    let n = candles.len();
    let closes: Vec<Option<Decimal>> = candles.iter().map(|c| Some(c.close)).collect();
    let highs: Vec<Option<Decimal>> = candles.iter().map(|c| Some(c.high)).collect();
    let lows: Vec<Option<Decimal>> = candles.iter().map(|c| Some(c.low)).collect();

    let rsi_values = rsi(&closes, cfg.rsi_period)?;

    let mut inverted_lows = Vec::with_capacity(n);
    for low in &lows {
        inverted_lows.push(match low {
            Some(v) => Some(neg(*v)?),
            None => None,
        });
    }

    let mut inverted_rsi = Vec::with_capacity(n);
    for value in &rsi_values {
        inverted_rsi.push(match value {
            Some(v) => Some(neg(*v)?),
            None => None,
        });
    }

    let price_peaks = find_peaks(&highs, cfg.rsi_order);
    let price_troughs = find_peaks(&inverted_lows, cfg.rsi_order);
    let rsi_peaks = find_peaks(&rsi_values, cfg.rsi_order);
    let rsi_troughs = find_peaks(&inverted_rsi, cfg.rsi_order);

    let mut regular_bullish = vec![false; n];
    let mut regular_bearish = vec![false; n];
    let mut hidden_bullish = vec![false; n];
    let mut hidden_bearish = vec![false; n];

    let matched_troughs = match_extrema(&price_troughs, &rsi_troughs, cfg.rsi_order);
    for pair in matched_troughs.windows(2) {
        let (prev_p, prev_r) = pair[0];
        let (curr_p, curr_r) = pair[1];
        let p_prev = lows[prev_p].unwrap_or(Decimal::ZERO);
        let p_curr = lows[curr_p].unwrap_or(Decimal::ZERO);
        let r_prev = rsi_values[prev_r].unwrap_or(Decimal::ZERO);
        let r_curr = rsi_values[curr_r].unwrap_or(Decimal::ZERO);
        if p_curr < p_prev && r_curr > r_prev {
            regular_bullish[curr_p] = true;
        } else if p_curr > p_prev && r_curr < r_prev {
            hidden_bullish[curr_p] = true;
        }
    }

    let matched_peaks = match_extrema(&price_peaks, &rsi_peaks, cfg.rsi_order);
    for pair in matched_peaks.windows(2) {
        let (prev_p, prev_r) = pair[0];
        let (curr_p, curr_r) = pair[1];
        let p_prev = highs[prev_p].unwrap_or(Decimal::ZERO);
        let p_curr = highs[curr_p].unwrap_or(Decimal::ZERO);
        let r_prev = rsi_values[prev_r].unwrap_or(Decimal::ZERO);
        let r_curr = rsi_values[curr_r].unwrap_or(Decimal::ZERO);
        if p_curr > p_prev && r_curr < r_prev {
            regular_bearish[curr_p] = true;
        } else if p_curr < p_prev && r_curr > r_prev {
            hidden_bearish[curr_p] = true;
        }
    }

    Ok(Divergences {
        rsi: rsi_values,
        regular_bullish,
        regular_bearish,
        hidden_bullish,
        hidden_bearish,
    })
}

pub fn match_extrema(price: &[usize], rsi_idx: &[usize], order: usize) -> Vec<(usize, usize)> {
    let mut matches = Vec::new();
    let mut used = vec![false; rsi_idx.len()];
    let mut prev_rsi: Option<usize> = None;

    for &p_idx in price {
        let mut best_distance = usize::MAX;
        let mut best_slot: Option<usize> = None;
        for (slot, &r_idx) in rsi_idx.iter().enumerate() {
            if used[slot] {
                continue;
            }
            if let Some(prev) = prev_rsi {
                if r_idx <= prev {
                    continue;
                }
            }
            let distance = p_idx.abs_diff(r_idx);
            if distance <= order && distance < best_distance {
                best_distance = distance;
                best_slot = Some(slot);
            }
        }
        if let Some(slot) = best_slot {
            let r_idx = rsi_idx[slot];
            used[slot] = true;
            matches.push((p_idx, r_idx));
            prev_rsi = Some(r_idx);
        }
    }
    matches
}
```

```rust
    pub index: usize,
    pub kind: DivergenceKind,
    pub confidence: Decimal,
}

pub fn price_diff(closes: &[Decimal]) -> Result<Decimal> {
    if closes.len() < 7 {
        return Ok(Decimal::ZERO);
    }
    let last = closes[closes.len() - 1];
    let past = closes[closes.len() - 7];
    if past == Decimal::ZERO {
        return Ok(Decimal::ZERO);
    }
    div(sub(last, past)?, past)
}

pub fn candidates(candles: &[Candle], cfg: &Config) -> Result<Vec<Candidate>> {
    let divs = detect(candles, cfg)?;
    let closes: Vec<Decimal> = candles.iter().map(|c| c.close).collect();
    let diff = price_diff(&closes)?;
    let latest_rsi = divs.rsi.last().copied().flatten();

    let mut regular_bullish = None;
    let mut regular_bearish = None;
    let mut hidden_bullish = None;
    let mut hidden_bearish = None;

    for idx in (0..divs.regular_bullish.len()).rev() {
        if regular_bullish.is_none() && divs.regular_bullish[idx] {
            regular_bullish = Some(idx);
        }
        if regular_bearish.is_none() && divs.regular_bearish[idx] {
            regular_bearish = Some(idx);
        }
        if hidden_bullish.is_none() && divs.hidden_bullish[idx] {
            hidden_bullish = Some(idx);
        }
        if hidden_bearish.is_none() && divs.hidden_bearish[idx] {
            hidden_bearish = Some(idx);
        }
        if regular_bullish.is_some()
            && regular_bearish.is_some()
            && hidden_bullish.is_some()
            && hidden_bearish.is_some()
        {
            break;
        }
    }

    let mut out = Vec::new();
    let tenth = Decimal::new(1, 1);
    let threshold = cfg.price_diff_threshold;

    if let Some(index) = regular_bullish {
        let mut confidence = if latest_rsi.is_some_and(|r| r < Decimal::from(30)) {
            Decimal::new(8, 1)
        } else {
            Decimal::new(6, 1)
        };
        if diff < neg(threshold)? {
            confidence = min(add(confidence, tenth)?, Decimal::ONE);
        }
        out.push(Candidate { index, kind: DivergenceKind::RegularBullish, confidence });
    }
    if let Some(index) = regular_bearish {
        let mut confidence = if latest_rsi.is_some_and(|r| r > Decimal::from(70)) {
            Decimal::new(8, 1)
        } else {
            Decimal::new(6, 1)
        };
        if diff > threshold {
            confidence = min(add(confidence, tenth)?, Decimal::ONE);
        }
        out.push(Candidate { index, kind: DivergenceKind::RegularBearish, confidence });
    }
    if let Some(index) = hidden_bullish {
        let mut confidence = Decimal::new(5, 1);
        if diff < neg(threshold)? {
            confidence = min(add(confidence, tenth)?, Decimal::ONE);
        }
        out.push(Candidate { index, kind: DivergenceKind::HiddenBullish, confidence });
    }
    if let Some(index) = hidden_bearish {
        let mut confidence = Decimal::new(5, 1);
        if diff > threshold {
            confidence = min(add(confidence, tenth)?, Decimal::ONE);
        }
        out.push(Candidate { index, kind: DivergenceKind::HiddenBearish, confidence });
    }

    Ok(out)
}

pub fn signals(candles: &[Candle], ticker: &str, cfg: &Config) -> Result<Option<Signal>> {
    if candles.is_empty() {
        return Ok(None);
    }
    let mut candidates = candidates(candles, cfg)?;
    if candidates.is_empty() {
        return Ok(None);
    }

    let last = candles.len() - 1;
    let latest_close = candles[last].close;

    if cfg.sma_enabled && candles.len() >= cfg.sma_period {
        let closes: Vec<Option<Decimal>> = candles.iter().map(|c| Some(c.close)).collect();
        let sma_series = sma(&closes, cfg.sma_period)?;
        let sma_value = sma_series[last];
        candidates.retain(|candidate| match (candidate.kind.action(), sma_value) {
            (Direction::Long, Some(value)) => latest_close > value,
            (Direction::Short, Some(value)) => latest_close < value,
            _ => false,
        });
        if candidates.is_empty() {
            return Ok(None);
        }
    }

    candidates.sort_by(|a, b| b.index.cmp(&a.index));
    let best = &candidates[0];
    let (sl_price, tp_price) = levels(candles, cfg, best.kind.action())?;
    let divergence_open_time_ms = candles[best.index].open_time_ms;
    let signal_candle_open_time_ms = candles[last].open_time_ms;

    Ok(Some(Signal {
        ticker: ticker.to_string(),
        action: best.kind.action(),
        confidence: best.confidence,
        entry_price: latest_close,
        reason: best.kind.reason().to_string(),
        sl_price,
        tp_price,
        divergence_index: best.index,
        divergence_kind: best.kind,
        divergence_open_time_ms,
        divergence_age_candles: last - best.index,
        signal_candle_open_time_ms,
    }))
}

fn levels(candles: &[Candle], cfg: &Config, action: Direction) -> Result<(Decimal, Decimal)> {
    use crate::strategy::indicators::atr::atr;

    let last = candles.len() - 1;
    let close = candles[last].close;
    if candles.len() < cfg.atr_period {
        return Ok((Decimal::ZERO, Decimal::ZERO));
    }
    let atr_series = atr(candles, cfg.atr_period)?;
    let atr_value = match atr_series[last] {
        Some(value) => value,
        None => return Ok((Decimal::ZERO, Decimal::ZERO)),
    };
    if atr_value <= Decimal::ZERO || close <= Decimal::ZERO {
        return Ok((Decimal::ZERO, Decimal::ZERO));
    }

    let atr_pct = mul(div(atr_value, close)?, Decimal::from(100))?;
    let mut sl_mult = cfg.atr_sl_multiplier;
    let mut tp_mult = cfg.atr_tp_multiplier;
    if atr_pct > cfg.atr_pct_high {
        sl_mult = mul(sl_mult, cfg.atr_high_adjust)?;
        tp_mult = mul(tp_mult, cfg.atr_high_adjust)?;
    }
    let sl_distance = mul(atr_value, sl_mult)?;
    let tp_distance = mul(atr_value, tp_mult)?;
    let (mut sl, tp) = match action {
        Direction::Long => (sub(close, sl_distance)?, add(close, tp_distance)?),
        Direction::Short => (add(close, sl_distance)?, sub(close, tp_distance)?),
    };

    if cfg.atr_sl_max_pct > Decimal::ZERO {
        match action {
            Direction::Long => {
                let max_sl = mul(close, sub(Decimal::ONE, cfg.atr_sl_max_pct)?)?;
                if sl < max_sl {
                    sl = max_sl;
                }
            }
            Direction::Short => {
                let max_sl = mul(close, add(Decimal::ONE, cfg.atr_sl_max_pct)?)?;
                if sl > max_sl {
                    sl = max_sl;
                }
            }
        }
    }

    Ok((sl, tp))
}
```

```python
import json

import os
import sys
from pathlib import Path

# CPU-dependent and not portable; pin the scalar path so tie behavior is reproducible and
# matches the Rust port of NumPy's scalar argsort ordering.
os.environ["NPY_DISABLE_CPU_FEATURES"] = (
    "AVX512F AVX512CD AVX512_SKX AVX512_CLX AVX512_CNL AVX512_ICL AVX2 FMA3 F16C"

    "hidden_bullish_divergence": "hidden_bullish",
    "hidden_bearish_divergence": "hidden_bearish",

def number(value):
    if value is None:
        return None
    try:
        result = float(value)
    except (TypeError, ValueError):
        return None
    if not math.isfinite(result):
        return None
    return repr(result)


def column(frame, *names):
    for name in names:
        if name in frame.columns:
            return frame[name]
    raise KeyError(f"none of {names} in {list(frame.columns)}")


def main() -> None:

    parser.add_argument("--symbol", required=True)
    parser.add_argument("--out", required=True)

    parser.add_argument("--atr-pct-high", type=float, default=5.0)
    parser.add_argument("--atr-high-adjust", type=float, default=1.5)
    parser.add_argument("--atr-sl-max-pct", type=float, default=0.0)
    args = parser.parse_args()

    raw = Path(args.data).read_bytes()
    frame = pd.read_csv(args.data)
    if "open_time_ms" in frame.columns:
        open_ms = [int(value) for value in frame["open_time_ms"]]
    elif "Timestamp" in frame.columns:
        frame["Timestamp"] = pd.to_datetime(frame["Timestamp"])
        open_ms = [int(value.timestamp() * 1000) for value in frame["Timestamp"]]
    else:
        raise KeyError("CSV needs open_time_ms or Timestamp")

    frame = frame.rename(
        columns={
            "open": "Open",
            "high": "High",
            "low": "Low",
            "close": "Close",
            "volume": "Volume",
        }
    )

    open_ = column(frame, "Open", "open")
    high = column(frame, "High", "high")
    low = column(frame, "Low", "low")
    close = column(frame, "Close", "close")

    candles = []
    for i in range(len(frame)):
        candles.append(

                "open_time_ms": open_ms[i],

    signals = []
    for end in range(args.window - 1, len(frame)):
        window = frame.iloc[end + 1 - args.window : end + 1]

            rsi_period=args.rsi_period,
            rsi_order=args.rsi_order,
            price_diff_threshold=args.price_diff_threshold,
            sma_enabled=args.sma_enabled,
            sma_period=args.sma_period,
            atr_period=args.atr_period,

            atr_tp_multiplier=args.atr_tp_multiplier,
            atr_pct_high=args.atr_pct_high,
            atr_high_adjust=args.atr_high_adjust,
            atr_sl_max_pct=args.atr_sl_max_pct,
        )
        if not produced:
            signals.append({"window_end": end, "signal": None})
            continue
        signal = produced[0]
        absolute_divergence_index = end + 1 - args.window + signal.divergence_idx
        signals.append(
            {
                "window_end": end,
                "signal": {
                    "action": "buy" if signal.action == "buy" else "sell",
                    "reason": signal.reason,
                    "kind": KIND[signal.reason],
                    "confidence": number(signal.confidence),
                    "entry": number(signal.entry_price),
                    "sl": number(signal.sl_price),
                    "tp": number(signal.tp_price),
                    "divergence_index": signal.divergence_idx,
                    "divergence_open_time_ms": open_ms[absolute_divergence_index],
                    "divergence_age_candles": args.window - 1 - signal.divergence_idx,
                    "signal_candle_open_time_ms": open_ms[end],
                },
            }
        )

    payload = {
        "symbol": args.symbol,
        "window": args.window,
        "source_sha256": hashlib.sha256(raw).hexdigest(),
        "versions": {
            "pandas": pd.__version__,

            "numpy": __import__("numpy").__version__,

            "rsi_period": args.rsi_period,
            "rsi_order": args.rsi_order,
            "price_diff_threshold": repr(args.price_diff_threshold),
            "sma_enabled": args.sma_enabled,
            "sma_period": args.sma_period,
            "atr_period": args.atr_period,
            "atr_sl_multiplier": repr(args.atr_sl_multiplier),
            "atr_tp_multiplier": repr(args.atr_tp_multiplier),
            "atr_pct_high": repr(args.atr_pct_high),
            "atr_high_adjust": repr(args.atr_high_adjust),
            "atr_sl_max_pct": repr(args.atr_sl_max_pct),
            "window": args.window,
        },
        "candles": candles,
        "indicators": {
            "rsi": [number(value) for value in rsi],
            "sma": [number(value) for value in sma],
            "atr": [number(value) for value in atr],

        "divergences": {

    print(f"wrote {args.out}: {len(candles)} candles, {positives} signals")
```

```rust
            rsi_order: self.config.rsi_order,
            price_diff_threshold: decimal_from_str(&self.config.price_diff_threshold).unwrap(),
            sma_enabled: self.config.sma_enabled,
            sma_period: self.config.sma_period,
            atr_period: self.config.atr_period,
            atr_sl_multiplier: decimal_from_str(&self.config.atr_sl_multiplier).unwrap(),
            atr_tp_multiplier: decimal_from_str(&self.config.atr_tp_multiplier).unwrap(),
            atr_pct_high: decimal_from_str(&self.config.atr_pct_high).unwrap(),
            atr_high_adjust: decimal_from_str(&self.config.atr_high_adjust).unwrap(),
            atr_sl_max_pct: decimal_from_str(&self.config.atr_sl_max_pct).unwrap(),

pub fn fixture_path(name: &str) -> PathBuf {
    PathBuf::from(env!("CARGO_MANIFEST_DIR"))
        .join("tests/parity")
        .join(format!("{name}.json"))
}

pub fn load_fixture(name: &str) -> Fixture {
    let path = fixture_path(name);
    let text = std::fs::read_to_string(&path)
        .unwrap_or_else(|e| panic!("read {}: {e}", path.display()));
    serde_json::from_str(&text).unwrap_or_else(|e| panic!("parse {}: {e}", path.display()))
}

pub fn close_enough(actual: Decimal, expected: Decimal, tolerance: f64) -> bool {
    if actual == expected {
        return true;
    }
    let diff = (actual - expected).abs();
    let scale = actual.abs().max(expected.abs()).max(Decimal::ONE);
    let tolerance = decimal_from_str(&format!("{tolerance:.9}")).unwrap();
    diff <= scale * tolerance
}

pub fn assert_close(actual: Decimal, expected: &str, tolerance: f64) {
    let expected = decimal_from_str(expected).unwrap();
    assert!(
        close_enough(actual, expected, tolerance),
        "got {actual}, want {expected}"
    );
}

pub fn assert_opt_close(actual: Option<Decimal>, expected: Option<&str>, tolerance: f64) {
    match (actual, expected) {
        (None, None) => {}
        (Some(a), Some(e)) => assert_close(a, e, tolerance),
        (a, e) => panic!("got {a:?}, want {e:?}"),
    }
}
```
